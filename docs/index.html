class Solution:
    def reverseString(self, s: List[str]) -> None:
        # In-place two-pointer O(1) extra memory
        l, r = 0, len(s) - 1
        while l < r:
            s[l], s[r] = s[r], s[l]
            l += 1
            r -= 1
